! --- QoS de voz en la WAN (parte 10) ---
! Clasifica por marca (EF/CS3, puesta por los switches) o por puertos de voz (equipos que no marcan)
ip access-list extended VOZ-RTP
 permit udp any any range 10000 20000
 permit udp any range 10000 20000 any
ip access-list extended VOZ-SIP
 permit udp any any eq 5060
 permit udp any eq 5060 any
class-map match-any CM-VOZ
 match dscp ef
 match access-group name VOZ-RTP
class-map match-any CM-SENAL
 match dscp cs3
 match access-group name VOZ-SIP
! Politica hija: LLQ 1 Mbps para voz, 256 kbps para senalizacion, WFQ para el resto
policy-map PM-WAN-HIJA
 class CM-VOZ
  priority 1000
 class CM-SENAL
  bandwidth 256
 class class-default
  fair-queue
! Politica padre: shaper a 10 Mbps (velocidad contratada simulada del enlace WAN)
policy-map PM-WAN-PADRE
 class class-default
  shape average 10000000
  service-policy PM-WAN-HIJA
! qos pre-classify: clasifica por puertos antes del cifrado IPsec
interface tunnel 1
 qos pre-classify
interface tunnel 2
 qos pre-classify
interface tunnel 3
 qos pre-classify
interface ethernet 0/0
 service-policy output PM-WAN-PADRE
 exit

! ============================================================
! VPN SITIO A SITIO HACIA AWS (SOC) - IKEv2 / IPsec
! R-TEMUCO (detras de NAT) <-> SRV-NOC strongSwan (EIP 98.83.92.138)
! Trafico protegido: 192.168.0.0/16 (sedes) <-> 10.40.0.0/16 (VPC)
! ============================================================
crypto ikev2 proposal PROP-AWS
 encryption aes-cbc-256
 integrity sha256
 group 14
 exit
crypto ikev2 policy POL-AWS
 proposal PROP-AWS
 exit
crypto ikev2 keyring KR-AWS
 peer SRV-NOC
  address 98.83.92.138
  pre-shared-key <clave no documentada>
  exit
 exit
crypto ikev2 profile PROF-AWS
 match identity remote address 98.83.92.138 255.255.255.255
 identity local fqdn r-temuco.agroprodel.cl
 authentication remote pre-share
 authentication local pre-share
 keyring local KR-AWS
 dpd 30 5 on-demand
 exit
crypto ipsec transform-set TS-AWS esp-aes 256 esp-sha256-hmac
 mode tunnel
 exit
ip access-list extended VPN-AWS
 permit ip 192.168.0.0 0.0.255.255 10.40.0.0 0.0.255.255
 exit
crypto map CM-AWS 10 ipsec-isakmp
 set peer 98.83.92.138
 set transform-set TS-AWS
 set pfs group14
 set ikev2-profile PROF-AWS
 match address VPN-AWS
 exit
interface ethernet 0/0
