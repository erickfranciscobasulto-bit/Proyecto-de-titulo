! --- IPsec fase 1 (IKE / ISAKMP) ---
crypto isakmp policy 10
 encryption aes 256
 hash sha256
 authentication pre-share
 group 14
 lifetime 86400
 exit
crypto isakmp key AgroTunel2026 address 203.0.113.6
crypto isakmp key AgroTunel2026 address 203.0.113.10

!
! --- IPsec fase 2 (transform-set y perfil) ---
crypto ipsec transform-set TS-AGRO esp-aes 256 esp-sha256-hmac
 mode transport
 exit
crypto ipsec profile PROF-AGRO
 set transform-set TS-AGRO
 exit
!

! --- QoS de voz en la WAN (parte 10) ---
! Clasifica por marca (EF/CS3, puesta por los switches) o por puertos de voz (equipos que no marcan)
ip access-list extended VOZ-RTP
 permit udp any any range 10000 20000
 permit udp any range 10000 20000 any
ip access-list extended VOZ-SIP
 permit udp any any eq 5060
 permit udp any eq 5060 any
class-map match-any CM-VOZ
 match dscp ef
 match access-group name VOZ-RTP
class-map match-any CM-SENAL
 match dscp cs3
 match access-group name VOZ-SIP
! Politica hija: LLQ 1 Mbps para voz, 256 kbps para senalizacion, WFQ para el resto
policy-map PM-WAN-HIJA
 class CM-VOZ
  priority 1000
 class CM-SENAL
  bandwidth 256
 class class-default
  fair-queue
! Politica padre: shaper a 10 Mbps (velocidad contratada simulada del enlace WAN)
policy-map PM-WAN-PADRE
 class class-default
  shape average 10000000
  service-policy PM-WAN-HIJA
! qos pre-classify: clasifica por puertos antes del cifrado IPsec
interface tunnel 1
 qos pre-classify
interface tunnel 2
 qos pre-classify
interface tunnel 3
 qos pre-classify
interface ethernet 0/0
 service-policy output PM-WAN-PADRE
 exit
!
end
