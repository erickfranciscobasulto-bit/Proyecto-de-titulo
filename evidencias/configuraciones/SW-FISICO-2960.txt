! ==========================================================
! SW-FISICO-TEM - Switch fisico Cisco Catalyst 2960 (rack)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Conectado al laboratorio por Gi0/1 (notebook -> br-rack -> SW-PUENTE e1/0)
! Estado: COMPLETA (partes 1 a 6) - QoS: mls qos, trust cisco-phone/cos en Fa0/4-5, trust dscp en Fa0/3 y Gi0/1 - Gi0/1 trunking nativa 999 y VLAN de acceso verificados
! ==========================================================
enable
configure terminal
hostname SW-FISICO-TEM
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP y VLANs ---
vtp mode transparent
vlan 10
 name TEM-SIEMBRA-RIEGO
vlan 50
 name TEM-VOZ
vlan 80
 name TEM-SERVIDORES
vlan 99
 name TEM-ADMINISTRACION
vlan 210
 name CON-DATOS
vlan 250
 name CON-VOZ
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
!
! --- QoS para voz: activacion global ---
mls qos
!
! --- Troncal Gi0/1 hacia el laboratorio (notebook / SW-PUENTE e1/0) ---
! El 2960 solo soporta 802.1Q: no requiere "switchport trunk encapsulation"
interface gigabitEthernet 0/1
 description TRONCAL-HACIA-NOTEBOOK-SW-PUENTE-e1/0
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,50,80,99,210,250,999
 mls qos trust dscp
 no shutdown
 exit
!
! --- Puertos de acceso ---
interface fastEthernet 0/2
 description PC-FISICO-TEMUCO
 switchport mode access
 switchport access vlan 10
 switchport voice vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface fastEthernet 0/3
 description SRV-ISSABEL-VLAN80
 switchport mode access
 switchport access vlan 80
 mls qos trust dscp
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface fastEthernet 0/4
 description PC-TEL-FISICO-CONCEPCION-V210-V250
 switchport mode access
 switchport access vlan 210
 switchport voice vlan 250
 mls qos trust device cisco-phone
 mls qos trust cos
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface fastEthernet 0/5
 description TELEFONO-FISICO-TEMUCO
 switchport mode access
 switchport access vlan 10
 switchport voice vlan 50
 mls qos trust device cisco-phone
 mls qos trust cos
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
!
! --- IP de gestion (VLAN 99) y LLDP para telefonos de otras marcas ---
interface vlan 99
 description GESTION-SW-FISICO-TEM
 ip address 192.168.40.245 255.255.255.240
 no shutdown
 exit
ip default-gateway 192.168.40.241
lldp run
!
end
write memory
