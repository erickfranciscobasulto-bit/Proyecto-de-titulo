! ==========================================================
! ACC1-TALCA - Switch de acceso 1 sucursal Talca (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Estado: EN PROGRESO (partes 1 a 2)
! ==========================================================
enable
configure terminal
hostname ACC1-TALCA
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP, VLANs y Rapid-PVST+ (root secundario) ---
vtp mode transparent
vlan 10
 name SIEMBRA-RIEGO
vlan 20
 name COSECHA-MOLIENDA
vlan 30
 name ENVASADO-BODEGA
vlan 40
 name VENTAS
vlan 50
 name VOZ
vlan 60
 name INVITADOS
vlan 70
 name IOT-RIEGO
vlan 99
 name ADMINISTRACION
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60,70,99,999 root secondary
!
! --- EtherChannel LACP: Po1 hacia DIST, Po3 hacia ACC2 ---
interface range ethernet 0/0 - 1
 description PO1-HACIA-DIST-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 1 mode active
 no shutdown
 exit
interface range ethernet 0/2 - 3
 description PO3-HACIA-ACC2-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 3 mode active
 no shutdown
 exit
interface port-channel 1
 description PO1-HACIA-DIST-TALCA
 exit
interface port-channel 3
 description PO3-HACIA-ACC2-TALCA
 exit
