! ==========================================================
! DIST-TALCA - Switch de distribucion sucursal Talca (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Estado: EN PROGRESO (partes 1 a 2)
! ==========================================================
enable
configure terminal
hostname DIST-TALCA
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP, VLANs y Rapid-PVST+ (root primario) ---
vtp mode transparent
vlan 10
 name SIEMBRA-RIEGO
vlan 20
 name COSECHA-MOLIENDA
vlan 30
 name ENVASADO-BODEGA
vlan 40
 name VENTAS
vlan 50
 name VOZ
vlan 60
 name INVITADOS
vlan 70
 name IOT-RIEGO
vlan 99
 name ADMINISTRACION
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60,70,99,999 root primary
!
! --- Troncal hacia R-TALCA e0/1 ---
interface ethernet 0/0
 description TRONCAL-HACIA-R-TALCA-e0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 no shutdown
 exit
!
! --- EtherChannel LACP: Po1 hacia ACC1, Po2 hacia ACC2 ---
interface range ethernet 0/1 - 2
 description PO1-HACIA-ACC1-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 1 mode active
 no shutdown
 exit
interface range ethernet 0/3 , ethernet 1/0
 description PO2-HACIA-ACC2-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 2 mode active
 no shutdown
 exit
interface port-channel 1
 description PO1-HACIA-ACC1-TALCA
 exit
interface port-channel 2
 description PO2-HACIA-ACC2-TALCA
 exit
