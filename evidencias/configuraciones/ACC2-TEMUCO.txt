! ==========================================================
! ACC2-TEMUCO - Switch de acceso 2 casa matriz (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Estado: EN PROGRESO (partes 1 a 3)
! ==========================================================
enable
configure terminal
hostname ACC2-TEMUCO
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP, VLANs y Rapid-PVST+ ---
vtp mode transparent
vlan 10
 name SIEMBRA-RIEGO
vlan 20
 name COSECHA-MOLIENDA
vlan 30
 name ENVASADO-BODEGA
vlan 40
 name VENTAS
vlan 50
 name VOZ
vlan 60
 name INVITADOS
vlan 70
 name IOT-RIEGO
vlan 80
 name SERVIDORES
vlan 99
 name ADMINISTRACION
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
!
! --- EtherChannel LACP: Po2 hacia DIST, Po3 hacia ACC1 ---
interface range ethernet 0/0 - 1
 description PO2-HACIA-DIST-TEMUCO
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,80,99,999
 channel-group 2 mode active
 no shutdown
 exit
interface range ethernet 0/2 - 3
 description PO3-HACIA-ACC1-TEMUCO
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,80,99,999
 channel-group 3 mode active
 no shutdown
 exit
interface port-channel 2
 description PO2-HACIA-DIST-TEMUCO
 exit
interface port-channel 3
 description PO3-HACIA-ACC1-TEMUCO
 exit
!
! --- Puertos de acceso (datos + voz) ---
interface ethernet 1/0
 description PC-V30-TEM-ENVASADO
 switchport mode access
 switchport access vlan 30
 switchport voice vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface ethernet 1/1
 description PC-V40-TEM-VENTAS
 switchport mode access
 switchport access vlan 40
 switchport voice vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
interface ethernet 1/2
 description PC-V10B-TEM-SIEMBRA
 switchport mode access
 switchport access vlan 10
 switchport voice vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
