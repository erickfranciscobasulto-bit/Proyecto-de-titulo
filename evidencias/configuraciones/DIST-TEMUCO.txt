! ==========================================================
! DIST-TEMUCO - Switch de distribucion casa matriz (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Estado: EN PROGRESO (partes 1 a 3)
! ==========================================================
enable
configure terminal
hostname DIST-TEMUCO
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP y VLANs ---
vtp mode transparent
vlan 10
 name SIEMBRA_RIEGO
vlan 20
 name COSECHA_MOLIENDA
vlan 30
 name ENVASADO_BODEGA
vlan 40
 name VENTAS
vlan 50
 name VOZ
vlan 60
 name INVITADOS
vlan 70
 name IOT_RIEGO
vlan 80
 name SERVIDORES
vlan 99
 name ADMINISTRACION
vlan 999
 name NATIVA
 exit
!
! --- Rapid-PVST+ (root primario) ---
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60,70,80,99,999 root primary
!
! --- Troncal hacia R-TEMUCO e0/1 ---
interface ethernet 0/0
 description TRONCAL-HACIA-R-TEMUCO-e0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,80,99,999
 no shutdown
 exit
