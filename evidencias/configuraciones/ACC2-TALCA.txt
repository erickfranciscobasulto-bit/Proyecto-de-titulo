! ==========================================================
! ACC2-TALCA - Switch de acceso 2 sucursal Talca (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Estado: EN PROGRESO (partes 1 a 2)
! ==========================================================
enable
configure terminal
hostname ACC2-TALCA
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP, VLANs y Rapid-PVST+ ---
vtp mode transparent
vlan 10
 name SIEMBRA-RIEGO
vlan 20
 name COSECHA-MOLIENDA
vlan 30
 name ENVASADO-BODEGA
vlan 40
 name VENTAS
vlan 50
 name VOZ
vlan 60
 name INVITADOS
vlan 70
 name IOT-RIEGO
vlan 99
 name ADMINISTRACION
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
!
! --- EtherChannel LACP: Po2 hacia DIST, Po3 hacia ACC1 ---
interface range ethernet 0/0 - 1
 description PO2-HACIA-DIST-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 2 mode active
 no shutdown
 exit
interface range ethernet 0/2 - 3
 description PO3-HACIA-ACC1-TALCA
 duplex full
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99,999
 channel-group 3 mode active
 no shutdown
 exit
interface port-channel 2
 description PO2-HACIA-DIST-TALCA
 exit
interface port-channel 3
 description PO3-HACIA-ACC1-TALCA
 exit
