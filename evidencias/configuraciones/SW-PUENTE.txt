! ==========================================================
! SW-PUENTE - Puente capa 2 entre rack fisico y laboratorio (IOL L2)
! Proyecto Agroprodel Ltda. - Configuracion aplicada
! Uso exclusivo de laboratorio: lleva el PC y telefono fisicos a Temuco
! y a Concepcion por un solo cable (VLAN 210 -> 10 y 250 -> 50 de Concepcion)
! Estado: EN PROGRESO (partes 1 a 3)
! ==========================================================
enable
configure terminal
hostname SW-PUENTE
no ip domain-lookup
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
!
! --- VTP, VLANs y Rapid-PVST+ ---
vtp mode transparent
vlan 10
 name TEM-SIEMBRA-RIEGO
vlan 50
 name TEM-VOZ
vlan 80
 name TEM-SERVIDORES
vlan 99
 name TEM-ADMINISTRACION
vlan 210
 name CON-DATOS
vlan 250
 name CON-VOZ
vlan 999
 name NATIVA
 exit
spanning-tree mode rapid-pvst
!
! --- Troncal hacia el rack fisico (Cloud / switch Cisco 2960) ---
interface ethernet 1/0
 description TRONCAL-HACIA-RACK-FISICO-2960
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,50,80,99,210,250,999
 no shutdown
 exit
!
! --- Troncal hacia DIST-TEMUCO e2/2 ---
interface ethernet 0/0
 description TRONCAL-HACIA-DIST-TEMUCO-e2/2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,50,80,99,999
 no shutdown
 exit
!
! --- Puertos hacia DIST-CONCEPCION (traduccion 210 -> 10 y 250 -> 50) ---
interface ethernet 0/1
 description HACIA-DIST-CONCEPCION-e2/2-DATOS-V210
 duplex full
 switchport mode access
 switchport access vlan 210
 spanning-tree bpdufilter enable
 no shutdown
 exit
interface ethernet 0/2
 description HACIA-DIST-CONCEPCION-e2/3-VOZ-V250
 duplex full
 switchport mode access
 switchport access vlan 250
 spanning-tree bpdufilter enable
 no shutdown
 exit
